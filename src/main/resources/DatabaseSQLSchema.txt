CREATE DATABASE IF NOT EXISTS aiApiDatabase;

USE aiApiDatabase;


-- =========================================
-- TABLES
-- =========================================

CREATE TABLE IF NOT EXISTS idea (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    idea TEXT NOT NULL
) ENGINE = InnoDB;


CREATE TABLE IF NOT EXISTS roast (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    roast TEXT NOT NULL,
    roast_score DECIMAL(3,1) NOT NULL,
    idea_id BIGINT NOT NULL,

    CONSTRAINT fk_roast_idea
        FOREIGN KEY (idea_id)
        REFERENCES idea(id)
        ON DELETE CASCADE
) ENGINE = InnoDB;


-- =========================================
-- ENSURE EXISTING COLUMNS ARE CORRECT
-- =========================================

ALTER TABLE idea
    MODIFY COLUMN idea TEXT NOT NULL;

ALTER TABLE roast
    MODIFY COLUMN roast TEXT NOT NULL,
    MODIFY COLUMN roast_score DECIMAL(3,1) NOT NULL;
